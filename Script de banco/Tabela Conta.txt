create table conta (
        id number(19,0) generated by default as identity,
	instituicao_financeira varchar2(255) not null,
        agencia varchar2(255) not null,
	numero_conta Varchar2(10) not null,
        ativo varchar2(1),
	cliente_id number(19,0) not null,
        

	CONSTRAINT PK_CONTA PRIMARY KEY (id),
	CONSTRAINT FK_CONTA_CLIENTE FOREIGN KEY (cliente_id) 
        REFERENCES cliente (id),
	CONSTRAINT UK_CONTA_BANCARIA_UNICA UNIQUE (instituicao_financeira, agencia, numero_conta)
    );
