create table cliente (
        id number(19,0) generated by default as identity,
        documento varchar2(14) not null,
        nome varchar2(255) not null,
        email varchar2(255) not null,
        telefone varchar2(255) not null,
        tipo_cliente varchar2(2) not null,
        data_nascimento date,
        nome_fantasia varchar2(255),
        data_cadastro timestamp default systimestamp not null,
        ativo varchar2(1) default 'S' not null,

        CONSTRAINT PK_CLIENTE PRIMARY KEY (id),
        CONSTRAINT UK_CLIENTE_DOCUMENTO UNIQUE (documento),
        CONSTRAINT CK_CLIENTE_ATIVO CHECK (ativo in ('S', 'N')),
        CONSTRAINT CK_CLIENTE_TIPO CHECK (tipo_cliente in ('PF', 'PJ'))
    );
