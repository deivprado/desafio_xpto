create table cliente (
        id number(19,0) generated by default as identity,
        documento varchar2(14 char) not null,
	nome varchar2(255 char) not null,
        email varchar2(255 char) not null,
        telefone varchar2(255 char) not null,
        tipo_cliente varchar2(2 char) not null,
	data_nascimento date,
	nome_fantasia varchar2(255 char),
        

	CONSTRAINT PK_CLIENTE PRIMARY KEY (id),
	CONSTRAINT UK_CLIENTE_DOCUMENTO UNIQUE (documento)
    );
