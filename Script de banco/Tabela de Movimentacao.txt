create table movimentacao (
    id number(19,0) generated by default as identity,
    tipo_movimentacao varchar2(10) not null,
    valor number(15,2) not null,
    data_movimentacao timestamp default systimestamp not null,
    descricao varchar2(255),
    conta_id number(19,0) not null,

    CONSTRAINT PK_MOVIMENTACAO PRIMARY KEY (id),
    CONSTRAINT FK_MOVIMENTACAO_CONTA FOREIGN KEY (conta_id) REFERENCES conta (id),
    CONSTRAINT CK_MOVIMENTACAO_TIPO CHECK (tipo_movimentacao in ('CREDITO', 'DEBITO')),
    CONSTRAINT CK_MOVIMENTACAO_VALOR CHECK (valor > 0)
);
